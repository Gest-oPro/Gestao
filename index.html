Entrada Rápida

Editar Data da Venda

Script SQL Supabase

Copie o código abaixo e cole no SQL Editor do seu painel Supabase. As regras RLS foram desativadas para permitir o fluxo completo.

CREATE TABLE IF NOT EXISTS products (
  id TEXT PRIMARY KEY, name TEXT NOT NULL, barcode TEXT, category TEXT,
  "costPrice" NUMERIC, "salePrice" NUMERIC, stock NUMERIC, "minStock" NUMERIC, description TEXT
);
ALTER TABLE products DISABLE ROW LEVEL SECURITY;

CREATE TABLE IF NOT EXISTS customers (
  id TEXT PRIMARY KEY, name TEXT NOT NULL, phone TEXT, email TEXT, address TEXT,
  credit NUMERIC, debt NUMERIC
);
ALTER TABLE customers DISABLE ROW LEVEL SECURITY;

CREATE TABLE IF NOT EXISTS sales (
  id TEXT PRIMARY KEY, date TEXT, seller TEXT, "customerId" TEXT, "customerName" TEXT, "customerPhone" TEXT,
  items JSONB, subtotal NUMERIC, discount NUMERIC, "creditUsed" NUMERIC, total NUMERIC, "totalCost" NUMERIC,
  "paymentMethod" TEXT, "receivedAmount" NUMERIC, "changeAmount" NUMERIC, "debtAmount" NUMERIC, "creditGenerated" NUMERIC, "dueDate" TEXT, status TEXT
);
ALTER TABLE sales DISABLE ROW LEVEL SECURITY;

CREATE TABLE IF NOT EXISTS settings (
  id TEXT PRIMARY KEY, data JSONB
);
ALTER TABLE settings DISABLE ROW LEVEL SECURITY;
